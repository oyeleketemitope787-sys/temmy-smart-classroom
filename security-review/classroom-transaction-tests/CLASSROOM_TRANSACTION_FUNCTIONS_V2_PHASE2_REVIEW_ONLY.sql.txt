BEGIN;
DO $guard$
BEGIN
 IF to_regprocedure('classroom_test._v2_claim_request(uuid,uuid,text,text)') IS NULL
 OR to_regprocedure('classroom_test._v2_lock_student_context(uuid,uuid)') IS NULL
 OR to_regclass('classroom_test.student_activity_records') IS NULL THEN
   RAISE EXCEPTION 'V2 helper foundation or fixture missing; aborting';
 END IF;
END $guard$;
SET ROLE temmy_ctest_data_owner;

CREATE OR REPLACE FUNCTION classroom_test.reverse_transaction(
 p_actor_teacher_id uuid,p_original_transaction_id uuid,p_request_id uuid)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog AS $fn$
DECLARE
 v_hash text; v_claim jsonb; v_original_student_id uuid;
 v_initial_class_id uuid;
 v_context jsonb; v_original classroom_test.point_transactions%ROWTYPE;
 v_balance bigint; v_reversal_id uuid; v_result jsonb; v_code text;
BEGIN
 PERFORM classroom_test._v2_assert_gateway_context();
 IF p_actor_teacher_id IS NULL OR p_request_id IS NULL
    OR p_original_transaction_id IS NULL THEN
   RAISE EXCEPTION USING ERRCODE='22004', MESSAGE='reversal_request_required';
 END IF;
 v_hash:=classroom_test._v2_canonical_hash(ARRAY[
   p_actor_teacher_id::text,p_original_transaction_id::text]);
 v_claim:=classroom_test._v2_claim_request(
   p_request_id,p_actor_teacher_id,'reverse_transaction',v_hash);
 IF NOT (v_claim->>'claimed')::boolean THEN
   RETURN (v_claim->'result') || jsonb_build_object('replayed',true);
 END IF;

 -- Discover the student without locking; the class is the first business row.
 SELECT pt.student_id INTO v_original_student_id
   FROM classroom_test.point_transactions pt
  WHERE pt.id=p_original_transaction_id;
 IF NOT FOUND THEN
   v_result:=jsonb_build_object('ok',false,'operation','reverse_transaction',
     'status','rejected','code','original_transaction_not_found','replayed',false);
   PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
     'reverse_transaction',NULL,'rejected',
     jsonb_build_object('code','original_transaction_not_found'));
   PERFORM classroom_test._v2_finish_request(p_request_id,'rejected',v_result);
   RETURN v_result;
 END IF;

 SELECT s.class_id INTO v_initial_class_id
   FROM classroom_test.students s WHERE s.id=v_original_student_id;
 IF NOT FOUND THEN
   RAISE EXCEPTION USING ERRCODE='40001', MESSAGE='original_student_membership_retry';
 END IF;
 -- The shared helper locks the current class before the student and
 -- re-reads membership; do not pre-lock the class here, which could acquire
 -- two class locks in an order that violates the global UUID ordering rule.
 v_context:=classroom_test._v2_lock_student_context(
   v_original_student_id,p_actor_teacher_id);
 IF v_context->>'status'='rejected' THEN
   v_code:=v_context->>'code';
   v_result:=jsonb_build_object('ok',false,'operation','reverse_transaction',
     'status','rejected','code',v_code,'replayed',false);
   PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
     'reverse_transaction',v_original_student_id,'rejected',
     jsonb_build_object('code',v_code));
   PERFORM classroom_test._v2_finish_request(p_request_id,'rejected',v_result);
   RETURN v_result;
 END IF;
 IF (v_context->>'class_id')::uuid IS DISTINCT FROM v_initial_class_id THEN
   RAISE EXCEPTION USING ERRCODE='40001', MESSAGE='original_student_class_changed_retry';
 END IF;

 SELECT pt.* INTO v_original FROM classroom_test.point_transactions pt
  WHERE pt.id=p_original_transaction_id FOR UPDATE;
 IF NOT FOUND OR v_original.student_id IS DISTINCT FROM v_original_student_id THEN
   RAISE EXCEPTION USING ERRCODE='40001', MESSAGE='original_transaction_changed_retry';
 END IF;

 v_code:=NULL;
 IF v_original.teacher_id IS DISTINCT FROM p_actor_teacher_id
    OR v_original.points<=0
    OR v_original.action_type<>'positive'
    OR v_original.action_name<>'Teacher award'
    OR v_original.reverses_transaction_id IS NOT NULL THEN
   v_code:='transaction_not_eligible_for_teacher_award_reversal';
 ELSIF EXISTS (SELECT 1 FROM classroom_test.point_transactions rev
                WHERE rev.reverses_transaction_id=v_original.id) THEN
   v_code:='transaction_already_reversed';
 END IF;
 v_balance:=(v_context->>'balance')::bigint;
 IF v_code IS NULL AND v_balance<v_original.points THEN
   v_code:='undo_insufficient_balance_full_reversal_required';
 END IF;
 IF v_code IS NOT NULL THEN
   v_result:=jsonb_build_object('ok',false,'operation','reverse_transaction',
     'status','rejected','code',v_code,'original_transaction_id',v_original.id,
     'replayed',false);
   PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
     'reverse_transaction',v_original_student_id,'rejected',
     jsonb_build_object('code',v_code,'original_transaction_id',v_original.id));
   PERFORM classroom_test._v2_finish_request(p_request_id,'rejected',v_result);
   RETURN v_result;
 END IF;

 INSERT INTO classroom_test.point_transactions
   (student_id,teacher_id,points,action_name,action_type,reason,source,
    reverses_transaction_id)
 VALUES (v_original.student_id,p_actor_teacher_id,-v_original.points,
   'Undo teacher award','undo','Full reversal of original award','teacher',
   v_original.id)
 RETURNING id INTO v_reversal_id;
 v_result:=jsonb_build_object('ok',true,'operation','reverse_transaction',
   'status','reversed','original_transaction_id',v_original.id,
   'reversal_transaction_id',v_reversal_id,'reversed_points',v_original.points,
   'balance',v_balance-v_original.points,'replayed',false);
 PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
   'reverse_transaction',v_original_student_id,'success',
   jsonb_build_object('original_transaction_id',v_original.id,
                      'reversal_transaction_id',v_reversal_id,
                      'points',v_original.points));
 PERFORM classroom_test._v2_finish_request(p_request_id,'completed',v_result);
 RETURN v_result;
END $fn$;

CREATE OR REPLACE FUNCTION classroom_test.score_activity(
 p_actor_teacher_id uuid,p_activity_record_id uuid,p_new_score integer,
 p_notes text,p_request_id uuid)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path=pg_catalog AS $fn$
DECLARE
 v_hash text; v_claim jsonb; v_student_id uuid;
 v_initial_class_id uuid;
 v_context jsonb; v_record classroom_test.student_activity_records%ROWTYPE;
 v_activity classroom_test.class_activities%ROWTYPE;
 v_balance bigint; v_old_score integer; v_delta integer;
 v_actual_delta integer; v_tx_id uuid; v_result jsonb; v_code text;
 v_notes text;
BEGIN
 PERFORM classroom_test._v2_assert_gateway_context();
 IF p_actor_teacher_id IS NULL OR p_activity_record_id IS NULL
    OR p_new_score IS NULL OR p_request_id IS NULL THEN
   RAISE EXCEPTION USING ERRCODE='22004', MESSAGE='activity_score_request_required';
 END IF;
 v_notes:=NULLIF(pg_catalog.btrim(p_notes),'');
 v_hash:=classroom_test._v2_canonical_hash(ARRAY[
   p_actor_teacher_id::text,p_activity_record_id::text,p_new_score::text,
   COALESCE(v_notes,'')]);
 v_claim:=classroom_test._v2_claim_request(
   p_request_id,p_actor_teacher_id,'score_activity',v_hash);
 IF NOT (v_claim->>'claimed')::boolean THEN
   RETURN (v_claim->'result') || jsonb_build_object('replayed',true);
 END IF;

 -- Preliminary reads only discover the class/student; no business row is locked.
 SELECT ar.student_id INTO v_student_id
   FROM classroom_test.student_activity_records ar
  WHERE ar.id=p_activity_record_id;
 IF NOT FOUND THEN
   v_result:=jsonb_build_object('ok',false,'operation','score_activity',
     'status','rejected','code','activity_record_not_found','replayed',false);
   PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
     'score_activity',NULL,'rejected',
     jsonb_build_object('code','activity_record_not_found'));
   PERFORM classroom_test._v2_finish_request(p_request_id,'rejected',v_result);
   RETURN v_result;
 END IF;
 SELECT s.class_id INTO v_initial_class_id
   FROM classroom_test.students s WHERE s.id=v_student_id;
 IF NOT FOUND THEN
   RAISE EXCEPTION USING ERRCODE='40001', MESSAGE='activity_student_membership_retry';
 END IF;
 -- Shared helper locks the current class before the student and
 -- re-reads membership. Avoid a preliminary class lock that could create
 -- an out-of-order multi-class lock if membership changed concurrently.
 v_context:=classroom_test._v2_lock_student_context(v_student_id,p_actor_teacher_id);
 IF v_context->>'status'='rejected' THEN
   v_code:=v_context->>'code';
   v_result:=jsonb_build_object('ok',false,'operation','score_activity',
     'status','rejected','code',v_code,'replayed',false);
   PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
     'score_activity',v_student_id,'rejected',jsonb_build_object('code',v_code));
   PERFORM classroom_test._v2_finish_request(p_request_id,'rejected',v_result);
   RETURN v_result;
 END IF;
 IF (v_context->>'class_id')::uuid IS DISTINCT FROM v_initial_class_id THEN
   RAISE EXCEPTION USING ERRCODE='40001', MESSAGE='activity_student_class_changed_retry';
 END IF;

 -- Operation-specific row order: activity record, then activity.
 SELECT ar.* INTO v_record FROM classroom_test.student_activity_records ar
  WHERE ar.id=p_activity_record_id FOR UPDATE;
 IF NOT FOUND OR v_record.student_id IS DISTINCT FROM v_student_id THEN
   RAISE EXCEPTION USING ERRCODE='40001', MESSAGE='activity_record_changed_retry';
 END IF;
 IF v_record.teacher_id IS DISTINCT FROM p_actor_teacher_id
    OR v_record.activity_id IS NULL THEN
   v_code:='activity_record_not_owned_or_not_scoreable';
 ELSE
   v_code:=NULL;
 END IF;

 IF v_code IS NULL THEN
   SELECT a.* INTO v_activity FROM classroom_test.class_activities a
    WHERE a.id=v_record.activity_id FOR SHARE;
   IF NOT FOUND THEN
     v_code:='activity_not_found';
   ELSIF NOT v_activity.active THEN
     v_code:='activity_inactive';
   ELSIF v_activity.class_id IS DISTINCT FROM v_initial_class_id THEN
     v_code:='activity_class_mismatch';
   END IF;
 END IF;
 IF v_code IS NULL AND (p_new_score<0 OR p_new_score>v_activity.max_score) THEN
   v_code:='score_out_of_activity_bounds';
 END IF;
 IF v_code IS NOT NULL THEN
   v_result:=jsonb_build_object('ok',false,'operation','score_activity',
     'status','rejected','code',v_code,'replayed',false);
   PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
     'score_activity',v_student_id,'rejected',jsonb_build_object('code',v_code));
   PERFORM classroom_test._v2_finish_request(p_request_id,'rejected',v_result);
   RETURN v_result;
 END IF;

 v_balance:=(v_context->>'balance')::bigint;
 v_old_score:=v_record.points_awarded;
 v_delta:=p_new_score-v_old_score;
 v_actual_delta:=v_delta;
 IF v_delta>0 THEN
   INSERT INTO classroom_test.point_transactions
    (student_id,teacher_id,points,action_name,action_type,reason,source)
   VALUES (v_student_id,p_actor_teacher_id,v_delta,
     'Activity score increase','activity_score','Activity score adjustment','teacher')
   RETURNING id INTO v_tx_id;
   v_balance:=v_balance+v_delta;
 ELSIF v_delta<0 THEN
   v_actual_delta:=-LEAST(ABS(v_delta)::bigint,GREATEST(v_balance,0))::integer;
   IF v_actual_delta<0 THEN
     INSERT INTO classroom_test.point_transactions
      (student_id,teacher_id,points,action_name,action_type,reason,source)
     VALUES (v_student_id,p_actor_teacher_id,v_actual_delta,
       'Activity score decrease','activity_score','Activity score adjustment','teacher')
     RETURNING id INTO v_tx_id;
     v_balance:=v_balance+v_actual_delta;
   ELSE
     v_actual_delta:=0;
   END IF;
 ELSE
   v_actual_delta:=0;
 END IF;

 UPDATE classroom_test.student_activity_records
    SET points_awarded=p_new_score,status='completed',
        completed_at=pg_catalog.now(),notes=v_notes
  WHERE id=p_activity_record_id;
 IF NOT FOUND THEN
   RAISE EXCEPTION USING ERRCODE='40001', MESSAGE='activity_record_update_retry';
 END IF;
 v_result:=jsonb_build_object('ok',true,'operation','score_activity',
   'status','scored','record_id',p_activity_record_id,
   'previous_score',v_old_score,'assigned_score',p_new_score,
   'requested_delta',v_delta,'actual_ledger_delta',v_actual_delta,
   'balance',GREATEST(v_balance,0),'transaction_id',v_tx_id,'replayed',false);
 PERFORM classroom_test._v2_write_audit(p_request_id,p_actor_teacher_id,
   'score_activity',v_student_id,'success',
   jsonb_build_object('record_id',p_activity_record_id,
     'previous_score',v_old_score,'assigned_score',p_new_score,
     'requested_delta',v_delta,'actual_ledger_delta',v_actual_delta));
 PERFORM classroom_test._v2_finish_request(p_request_id,'completed',v_result);
 RETURN v_result;
END $fn$;

REVOKE ALL ON FUNCTION classroom_test.reverse_transaction(uuid,uuid,uuid) FROM PUBLIC;
REVOKE ALL ON FUNCTION classroom_test.score_activity(uuid,uuid,integer,text,uuid) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION classroom_test.reverse_transaction(uuid,uuid,uuid)
 TO temmy_ctest_gateway;
GRANT EXECUTE ON FUNCTION classroom_test.score_activity(uuid,uuid,integer,text,uuid)
 TO temmy_ctest_gateway;
RESET ROLE;
COMMIT;
