BEGIN;

-- anon: schema/table access must not permit direct ledger writes.
SET LOCAL ROLE temmy_ctest_anon;
DO $test$
DECLARE v_denied boolean:=false;
BEGIN
 BEGIN
  INSERT INTO classroom_test.point_transactions
   (student_id,teacher_id,points,action_name,action_type,source)
  VALUES
   ('30000000-0000-4000-8000-000000000001',
    '10000000-0000-4000-8000-000000000001',
    1,'ACL probe','positive','test');
 EXCEPTION WHEN insufficient_privilege THEN
  v_denied:=true;
 END;
 IF NOT v_denied THEN RAISE EXCEPTION 'anon unexpectedly inserted a ledger row'; END IF;
END $test$;

-- authenticated: SELECT is intentionally allowed on selected reference
-- tables, but direct protected writes must still be denied.
RESET ROLE;
SET LOCAL ROLE temmy_ctest_authenticated;
DO $test$
DECLARE v_denied boolean:=false;
BEGIN
 BEGIN
  UPDATE classroom_test.students
     SET active=false
   WHERE id='30000000-0000-4000-8000-000000000001';
 EXCEPTION WHEN insufficient_privilege THEN
  v_denied:=true;
 END;
 IF NOT v_denied THEN RAISE EXCEPTION 'authenticated unexpectedly updated a student'; END IF;
END $test$;

-- Gateway: business entrypoint execution is its only write path; no direct
-- table DML is granted to the gateway role.
RESET ROLE;
SET LOCAL ROLE temmy_ctest_gateway;
DO $test$
DECLARE v_denied boolean:=false;
BEGIN
 BEGIN
  DELETE FROM classroom_test.point_transactions
   WHERE id='60000000-0000-4000-8000-000000000001';
 EXCEPTION WHEN insufficient_privilege THEN
  v_denied:=true;
 END;
 IF NOT v_denied THEN RAISE EXCEPTION 'gateway unexpectedly deleted a ledger row'; END IF;
END $test$;

RESET ROLE;
ROLLBACK;
