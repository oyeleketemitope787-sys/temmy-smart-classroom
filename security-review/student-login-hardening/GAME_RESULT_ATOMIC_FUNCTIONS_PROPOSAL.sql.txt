CREATE OR REPLACE FUNCTION public.record_game_attempt_answer(
    p_session_token uuid,
    p_game_id text,
    p_question_index smallint,
    p_choice_index smallint
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
    v_session record;
    v_attempt record;
    v_question record;
    v_next_question record;
    v_rule record;
    v_is_correct boolean;
    v_next_streak smallint;
    v_next_index smallint;
    v_next_correct_count smallint;
    v_points_delta integer := 0;
    v_new_score integer;
    v_transaction_id uuid;
    v_updated_session_id uuid;
BEGIN
    IF p_session_token IS NULL
       OR p_game_id IS NULL
       OR p_question_index IS NULL
       OR p_choice_index IS NULL
       OR p_question_index < 0
       OR p_choice_index NOT BETWEEN 0 AND 3 THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'invalid_answer');
    END IF;

    SELECT
        sgs.id AS session_id,
        sgs.student_id,
        sgs.class_id,
        sgs.game_id,
        sgs.expires_at,
        sgs.used_at,
        c.teacher_id,
        c.active AS class_active,
        st.active AS student_active,
        st.class_id AS student_class_id
    INTO v_session
    FROM public.student_game_sessions AS sgs
    JOIN public.classes AS c ON c.id = sgs.class_id
    JOIN public.students AS st ON st.id = sgs.student_id
    WHERE sgs.session_token = p_session_token
    FOR UPDATE OF sgs;

    IF NOT FOUND
       OR v_session.game_id IS DISTINCT FROM p_game_id
       OR v_session.student_class_id IS DISTINCT FROM v_session.class_id THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'invalid_session');
    END IF;

    SELECT a.*
    INTO v_attempt
    FROM public.student_game_attempts AS a
    WHERE a.session_id = v_session.session_id
    FOR UPDATE;

    IF NOT FOUND OR v_attempt.game_id IS DISTINCT FROM v_session.game_id THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'invalid_attempt');
    END IF;

    SELECT q.*
    INTO v_question
    FROM public.student_game_attempt_questions AS q
    WHERE q.attempt_id = v_attempt.id
      AND q.question_index = p_question_index
    FOR UPDATE;

    IF NOT FOUND THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'invalid_question');
    END IF;

    -- Idempotent retry: the exact same answer returns stored state and cannot
    -- increment score or create another point transaction.
    IF v_question.submitted_choice_index IS NOT NULL THEN
        IF v_question.submitted_choice_index <> p_choice_index THEN
            RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'answer_conflict');
        END IF;

        IF v_attempt.status = 'completed' THEN
            RETURN pg_catalog.jsonb_build_object(
                'ok', true,
                'status', 'completed',
                'duplicate', true,
                'game_id', v_attempt.game_id,
                'score', v_attempt.verified_score,
                'max_score', v_attempt.max_score,
                'points_awarded', v_attempt.points_awarded,
                'score_rule_version', v_attempt.score_rule_version
            );
        END IF;

        IF v_attempt.status <> 'in_progress'
           OR v_session.used_at IS NOT NULL
           OR v_session.expires_at <= pg_catalog.now()
           OR v_session.class_active IS DISTINCT FROM true
           OR v_session.student_active IS DISTINCT FROM true THEN
            RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'invalid_session');
        END IF;

        SELECT q.*
        INTO v_next_question
        FROM public.student_game_attempt_questions AS q
        WHERE q.attempt_id = v_attempt.id
          AND q.question_index = v_attempt.next_question_index;

        IF NOT FOUND THEN
            RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'attempt_state_invalid');
        END IF;

        RETURN pg_catalog.jsonb_build_object(
            'ok', true,
            'status', 'in_progress',
            'duplicate', true,
            'game_id', v_attempt.game_id,
            'question_count', v_attempt.question_count,
            'score', v_attempt.verified_score,
            'max_score', v_attempt.max_score,
            'question', pg_catalog.jsonb_build_object(
                'question_index', v_next_question.question_index,
                'prompt', v_next_question.prompt,
                'options', v_next_question.options,
                'hint', v_next_question.hint
            )
        );
    END IF;

    IF v_attempt.status <> 'in_progress'
       OR v_session.used_at IS NOT NULL
       OR v_session.expires_at <= pg_catalog.now()
       OR v_session.class_active IS DISTINCT FROM true
       OR v_session.student_active IS DISTINCT FROM true THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'invalid_session');
    END IF;

    IF p_question_index <> v_attempt.next_question_index THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'unexpected_question');
    END IF;

    SELECT r.*
    INTO v_rule
    FROM public.game_scoring_catalogue AS r
    WHERE r.game_id = v_attempt.game_id
      AND r.rule_version = v_attempt.score_rule_version;

    IF NOT FOUND
       OR v_rule.question_count <> v_attempt.question_count
       OR v_rule.max_score <> v_attempt.max_score
       OR (v_rule.rule_version = 'number-challenge-streak-v1'
           AND (v_rule.question_count <> 15 OR v_rule.max_score <> 41))
       OR (v_rule.rule_version = 'one-point-per-correct-v1'
           AND v_rule.max_score <> v_rule.question_count)
       OR v_rule.rule_version NOT IN (
            'number-challenge-streak-v1',
            'one-point-per-correct-v1'
       ) THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'scoring_rule_unavailable');
    END IF;

    v_is_correct := p_choice_index = v_question.correct_choice_index;
    v_next_streak := CASE WHEN v_is_correct THEN v_attempt.current_streak + 1 ELSE 0 END;
    v_next_correct_count := v_attempt.correct_count + CASE WHEN v_is_correct THEN 1 ELSE 0 END;

    IF v_attempt.score_rule_version = 'number-challenge-streak-v1' THEN
        IF v_is_correct THEN
            v_points_delta := 1;
            IF v_next_streak >= 3 THEN
                v_points_delta := v_points_delta + 1;
            END IF;
            IF v_next_streak >= 5 THEN
                v_points_delta := v_points_delta + 1;
            END IF;
            IF p_question_index = v_attempt.question_count - 1 THEN
                v_points_delta := v_points_delta + 2;
            END IF;
        END IF;
    ELSIF v_attempt.score_rule_version = 'one-point-per-correct-v1' THEN
        IF v_is_correct THEN
            v_points_delta := 1;
        END IF;
    ELSE
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'unsupported_scoring_rule');
    END IF;

    v_new_score := v_attempt.verified_score + v_points_delta;
    v_next_index := v_attempt.next_question_index + 1;

    IF v_new_score < 0 OR v_new_score > v_attempt.max_score
       OR v_next_correct_count > v_attempt.question_count THEN
        RETURN pg_catalog.jsonb_build_object('ok', false, 'error', 'scoring_rule_violation');
    END IF;

    UPDATE public.student_game_attempt_questions
    SET submitted_choice_index = p_choice_index,
        is_correct = v_is_correct,
        answered_at = pg_catalog.now()
    WHERE attempt_id = v_attempt.id
      AND question_index = p_question_index
      AND submitted_choice_index IS NULL;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Game answer state changed unexpectedly';
    END IF;

    IF v_next_index = v_attempt.question_count THEN
        INSERT INTO public.point_transactions (
            student_id,
            teacher_id,
            points,
            action_name,
            action_type,
            reason,
            source
        )
        VALUES (
            v_session.student_id,
            v_session.teacher_id,
            v_new_score,
            v_rule.display_name,
            'activity_score',
            'Game completed — ' || v_new_score || '/' || v_attempt.max_score,
            'game'
        )
        RETURNING id INTO v_transaction_id;

        UPDATE public.student_game_attempts
        SET status = 'completed',
            next_question_index = v_next_index,
            correct_count = v_next_correct_count,
            verified_score = v_new_score,
            current_streak = v_next_streak,
            completed_at = pg_catalog.now(),
            points_awarded = v_new_score,
            point_transaction_id = v_transaction_id
        WHERE id = v_attempt.id
          AND status = 'in_progress';

        IF NOT FOUND THEN
            RAISE EXCEPTION 'Game attempt completion changed unexpectedly';
        END IF;

        UPDATE public.student_game_sessions
        SET used_at = pg_catalog.now()
        WHERE id = v_session.session_id
          AND used_at IS NULL
        RETURNING id INTO v_updated_session_id;

        IF v_updated_session_id IS NULL THEN
            RAISE EXCEPTION 'Game session consumption failed';
        END IF;

        RETURN pg_catalog.jsonb_build_object(
            'ok', true,
            'status', 'completed',
            'game_id', v_attempt.game_id,
            'score', v_new_score,
            'max_score', v_attempt.max_score,
            'points_awarded', v_new_score,
            'score_rule_version', v_attempt.score_rule_version,
            'correct_count', v_next_correct_count,
            'question_count', v_attempt.question_count
        );
    END IF;

    UPDATE public.student_game_attempts
    SET next_question_index = v_next_index,
        correct_count = v_next_correct_count,
        verified_score = v_new_score,
        current_streak = v_next_streak
    WHERE id = v_attempt.id
      AND status = 'in_progress';

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Game attempt state changed unexpectedly';
    END IF;

    SELECT q.*
    INTO v_next_question
    FROM public.student_game_attempt_questions AS q
    WHERE q.attempt_id = v_attempt.id
      AND q.question_index = v_next_index;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Next game question is missing';
    END IF;

    RETURN pg_catalog.jsonb_build_object(
        'ok', true,
        'status', 'in_progress',
        'game_id', v_attempt.game_id,
        'question_count', v_attempt.question_count,
        'answer_correct', v_is_correct,
        'points_delta', v_points_delta,
        'score', v_new_score,
        'max_score', v_attempt.max_score,
        'question', pg_catalog.jsonb_build_object(
            'question_index', v_next_question.question_index,
            'prompt', v_next_question.prompt,
            'options', v_next_question.options,
            'hint', v_next_question.hint
        )
    );
END;
$function$;
